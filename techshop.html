-- Trigger #28: Tự động chuyển trạng thái khi thanh toán
CREATE OR REPLACE FUNCTION auto_update_order_status()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.payment_status = 'Completed' THEN
    UPDATE orders
    SET delivery_status = 'Shipping'
    WHERE order_id = NEW.order_id
      AND delivery_status = 'Pending';
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

SQL Query – Phân hạng khách hàng (Query #5)SELECT
-- Xếp hạng khách hàng theo chi tiêu
WITH CustomerStats AS (
  SELECT u.user_id, u.full_name,
         COUNT(o.order_id) AS total_orders,
         SUM(o.total_amount) AS lifetime_value
  FROM users u
  JOIN accounts a ON u.user_id = a.user_id
  JOIN orders o ON a.account_id = o.account_id
  WHERE o.delivery_status = 'delivered'
  GROUP BY u.user_id, u.full_name
)
SELECT *, CASE
  WHEN lifetime_value >= 50000000 THEN 'S-VIP'
  WHEN lifetime_value >= 20000000 THEN 'S-Member'
  ELSE 'S-New'
END AS membership_tier
FROM CustomerStats
ORDER BY lifetime_value DESC;

CREATE OR REPLACE FUNCTION notify_low_stock()
RETURNS TRIGGER AS $$
BEGIN
  IF NEW.stock_quantity < 2 THEN
    RAISE NOTICE 'Cảnh báo: SP % sắp hết!', NEW.name;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Gọi: SELECT get_account_total_spend(1);
CREATE OR REPLACE FUNCTION get_account_total_spend(p_account_id INT)
RETURNS DECIMAL AS $$
DECLARE total DECIMAL;
BEGIN
  SELECT SUM(total_amount) INTO total
  FROM orders
  WHERE account_id = p_account_id
    AND delivery_status = 'delivered';
  RETURN COALESCE(total, 0);
END;
$$ LANGUAGE plpgsql;
